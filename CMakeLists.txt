# swarm_package(<name> DEPS <targets>): library from src/, public include/, tests/ if any.
function(swarm_package name)
  cmake_parse_arguments(P "" "" "DEPS" ${ARGN})
  file(GLOB srcs CONFIGURE_DEPENDS src/*.cpp)
  add_library(${name} STATIC ${srcs})
  target_include_directories(${name} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
  install(TARGETS ${name} ARCHIVE DESTINATION lib)
  install(DIRECTORY include/ DESTINATION include)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/package.xml")
    install(FILES package.xml DESTINATION share/${name})
  endif()
  target_link_libraries(${name} PUBLIC ${P_DEPS})
  target_compile_options(${name} PRIVATE -Wall -Wextra)
  file(GLOB tests CONFIGURE_DEPENDS tests/*.cpp)
  if(SWARM_TESTS AND tests)
    add_executable(${name}_tests ${tests})
    target_link_libraries(${name}_tests PRIVATE ${name} GTest::gtest_main)
    target_compile_definitions(${name}_tests PRIVATE SWARM_ROOT="${PROJECT_SOURCE_DIR}")
    gtest_discover_tests(${name}_tests)
  endif()
endfunction()
